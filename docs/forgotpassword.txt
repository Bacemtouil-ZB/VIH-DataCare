🟢 ÉTAPE 5 — Forgot Password Controller
🎯 Rôle :

vérifier email

générer token

sauvegarder token

envoyer email

import crypto from "crypto";
import { transporter } from "../services/emailService.js";
import { findUserByEmail, saveResetToken } from "../models/userModel.js";

export const forgotPassword = async (req, res) => {
  const { email } = req.body;

  try {
    const user = await findUserByEmail(email);
    if (!user) {
      return res.status(404).json({ message: "Email not found" });
    }

    // 1️⃣ Generate secure token
    const resetToken = crypto.randomBytes(32).toString("hex");

    // 2️⃣ Expiration time (1 hour)
    const expiration = new Date(Date.now() + 3600000);

    // 3️⃣ Save in DB
    await saveResetToken(user.id, resetToken, expiration);

    // 4️⃣ Send email
    const resetLink = `http://localhost:5173/reset-password/${resetToken}`;

    await transporter.sendMail({
      from: '"VihDataCare" <noreply@vihdatacare.com>',
      to: email,
      subject: "Reset Your Password",
      html: `
        <h3>Password Reset</h3>
        <p>Click the link below to reset your password:</p>
        <a href="${resetLink}">${resetLink}</a>
        <p>This link expires in 1 hour.</p>
      `,
    });

    res.status(200).json({ message: "Reset email sent" });

  } catch (error) {
    res.status(500).json({ message: "Server error" });
  }
};

🟢 ÉTAPE 7 — Reset Password Controller
🎯 Rôle :

vérifier token

vérifier expiration

hasher nouveau password

supprimer token

import bcrypt from "bcrypt";

export const resetPassword = async (req, res) => {
  const { token } = req.params;
  const { newPassword } = req.body;

  try {
    const result = await pool.query(
      `SELECT * FROM users 
       WHERE reset_token = $1 
       AND reset_token_expires > NOW()`,
      [token]
    );

    if (result.rows.length === 0) {
      return res.status(400).json({ message: "Invalid or expired token" });
    }

    const user = result.rows[0];

    const hashedPassword = await bcrypt.hash(newPassword, 10);

    await pool.query(
      `UPDATE users
       SET password = $1,
           reset_token = NULL,
           reset_token_expires = NULL
       WHERE id = $2`,
      [hashedPassword, user.id]
    );

    res.status(200).json({ message: "Password reset successful" });

  } catch (error) {
    res.status(500).json({ message: "Server error" });
  }
};

🔐 Sécurité importante

✔ Toujours hasher password
✔ Toujours supprimer reset_token après usage
✔ Toujours mettre expiration
✔ Ne jamais renvoyer si email existe ou pas en production (anti-enumeration attack)

🧠 Pourquoi token généré seulement dans forgotPassword ?

Parce que :

forgotPassword = demande de reset
resetPassword = validation + update

ResetPassword ne génère rien. Il vérifie.
